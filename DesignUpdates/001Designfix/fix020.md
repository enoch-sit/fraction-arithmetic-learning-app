3. **Implement Core Functionality**:
   ```typescript
   // In src/apps/integerparts/App.tsx
   
   import React, { useState } from 'react';
   import { AppHeader } from '@/shared/components/AppHeader';
   import { FracInputStepper } from '@/shared/components/FracInputStepper';
   import { FractionBar } from '@/shared/components/FractionBar';
   import './app.css';
   
   function IntegerPartsApp() {
     const [numerator, setNumerator] = useState(7);
     const [denominator, setDenominator] = useState(3);
     
     // Calculate integer and fractional parts
     const integerPart = Math.floor(numerator / denominator);
     const remainderPart = numerator % denominator;
     
     // Check if improper fraction
     const isImproper = numerator >= denominator;
     
     return (
       <div className="integer-parts-app">
         <AppHeader title="整數的部份" />
         
         <div className="content">
           <div className="input-section">
             <h3>輸入分數</h3>
             <div className="fraction-input">
               <FracInputStepper
                 label="分子"
                 value={numerator}
                 onChange={setNumerator}
                 min={1}
                 max={99}
               />
               <span className="fraction-line">/</span>
               <FracInputStepper
                 label="分母"
                 value={denominator}
                 onChange={setDenominator}
                 min={1}
                 max={20}
               />
             </div>
           </div>
           
           <div className="visualization-section">
             <h3>視覺化顯示</h3>
             <div className="fraction-bars">
               {/* Show multiple whole bars for integer part */}
               {Array.from({ length: integerPart }).map((_, i) => (
                 <div key={`whole-${i}`} className="whole-bar">
                   <FractionBar
                     numerator={denominator}
                     denominator={denominator}
                     color="#4CAF50"
                   />
                   <span className="bar-label">1個整數</span>
                 </div>
               ))}
               
               {/* Show partial bar for remainder */}
               {remainderPart > 0 && (
                 <div className="partial-bar">
                   <FractionBar
                     numerator={remainderPart}
                     denominator={denominator}
                     color="#FF9800"
                   />
                   <span className="bar-label">
                     剩餘 {remainderPart}/{denominator}
                   </span>
                 </div>
               )}
             </div>
           </div>
           
           <div className="result-section">
             <h3>分析結果</h3>
             <div className="analysis">
               <div className="fraction-display">
                 <span className="original">
                   原分數: <strong>{numerator}/{denominator}</strong>
                 </span>
                 {isImproper && (
                   <span className="improper-notice">
                     (假分數)
                   </span>
                 )}
               </div>
               
               <div className="breakdown">
                 <div className="integer-part">
                   <span className="label">整數部分:</span>
                   <span className="value">{integerPart}</span>
                 </div>
                 
                 <div className="fractional-part">
                   <span className="label">分數部分:</span>
                   <span className="value">
                     {remainderPart}/{denominator}
                   </span>
                 </div>
                 
                 <div className="mixed-number">
                   <span className="label">帶分數:</span>
                   <span className="value">
                     {integerPart > 0 && `${integerPart} `}
                     {remainderPart > 0 && `${remainderPart}/${denominator}`}
                     {integerPart === 0 && remainderPart === 0 && '0'}
                   </span>
                 </div>
               </div>
               
               <div className="equation">
                 {numerator}/{denominator} = {integerPart} + {remainderPart}/{denominator}
                 {integerPart > 0 && remainderPart > 0 && (
                   <span> = {integerPart} {remainderPart}/{denominator}</span>
                 )}
               </div>
             </div>
           </div>
         </div>
       </div>
     );
   }
   
   export default IntegerPartsApp;
   ```
